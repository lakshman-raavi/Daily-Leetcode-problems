1. Find the XOR of Numbers Which Appear Twice  (3158)

Solution : class Solution {
    public int duplicateNumbersXOR(int[] nums) {

        Map<Integer,Integer> map=new HashMap<>();
        for(int i : nums){
            map.put(i,map.getOrDefault(i,0)+1);
        }


        int xor=0;
        List<Integer> list=new ArrayList<>();
        for(Map.Entry<Integer,Integer> e : map.entrySet()){
            int val=e.getValue();
            if(val==2){
                xor^=e.getKey();
            }
        }
        return xor;
    }
}


2. Find The Original Array of Prefix Xor (2433)

Solution : class Solution {
    public int[] findArray(int[] pref) {
        int n=pref.length;
        int[] res=new int[n];
        int xor=0;
        res[0]=pref[0];
        for(int i=1;i<n;i++){
            xor=pref[i]^pref[i-1];
            res[i]=xor;
        }
        return res;
    }
}

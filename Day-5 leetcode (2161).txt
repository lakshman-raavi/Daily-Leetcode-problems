1. Partition Array According to Given Pivot  (2161)

Solution:  class Solution {
    public int[] pivotArray(int[] nums, int pivot) {
        int n=nums.length;
        int lessc=0;
        int greatc=0;
        int equalc=0;
        for(int i=0;i<n;i++){
            if(nums[i]<pivot){
                lessc++;
            }
            else if(nums[i]>pivot){
                greatc++;
            }
            else{
                equalc++;
            }
        }
        int[] left=new int[lessc];
        int[] right=new int[greatc];
        int k1=0,k2=0;
        for(int i=0;i<n;i++){
            if(nums[i]<pivot){
                left[k1]=nums[i];
                k1++;
            }
            else if(nums[i]>pivot){
                right[k2]=nums[i];
                k2++;
            }
        }
        int k=0;
        for(int i=0;i<k1;i++){
            nums[k]=left[i];
            k++;
        }
        for(int i=0;i<equalc;i++){
            nums[k]=pivot;
            k++;
        }
        for(int i=0;i<k2;i++){
            nums[k]=right[i];
            k++;
        }
        return nums;
    }
}

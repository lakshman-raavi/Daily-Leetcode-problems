1.  Relative Ranks  (506)

Solution :  class Solution {
    public String[] findRelativeRanks(int[] score) {
        int n=score.length;
        int[] duplicate=new int[n];
        for(int i=0;i<n;i++){
            duplicate[i]=score[i];
        }
        Arrays.sort(score);
        Map<Integer,String> map=new HashMap<>();
        int rank=n;
        for(int num : score){
            if(rank==1){
            map.put(num,"Gold Medal");
            }
            else if(rank==2){
            map.put(num,"Silver Medal");
            }
            else if(rank==3){
            map.put(num,"Bronze Medal");
            }
            else{
                map.put(num,Integer.toString(rank));
            }
            rank--;
        }

        String[] ans =new String[n];
        for(int i=0;i<n;i++){
            ans[i]=map.get(duplicate[i]);
        }
        
        return ans;
    }
}


2.  Longest Common Prefix  (14)

Solution :  class Solution {
    public String longestCommonPrefix(String[] strs) {
        int len = Integer.MAX_VALUE;

        if (strs.length == 1) {
            return strs[0];
        }

        for (String word : strs) {
            len = Math.min(word.length(), len);
        }
        boolean is = true;
        StringBuilder res = new StringBuilder();
        for (int i = 0; i < len; i++) {
            String copy = strs[0].substring(0, i + 1);
            for (String s : strs) {
                if (s.length() >= copy.length()) {
                    if (s.substring(0, i + 1).equals(copy)) {
                        is = true;
                    } else {
                        is = false;
                        break;
                    }
                }
            }
            if (is) {
                res.setLength(0);
                res.append(copy);
            } else {
                break;
            }
        }

        return res.toString();

    }
}

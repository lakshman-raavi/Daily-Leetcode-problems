1. Find the Prefix Common Array of Two Arrays  (2567)

Solution : class Solution {
    public int[] findThePrefixCommonArray(int[] A, int[] B) {
        int count=0;
        int n=A.length;
        int[] c=new int[n];

        
            if(A[0]==B[0]){
                c[0]=1;
            }
            else{
                c[0]=0;
            }
            
        for(int i=0;i<n;i++){
            count=0;
            for(int k=i;k>=0;k--){
                for(int j=i;j>=0;j--){
                    if(A[k]==B[j]){
                        count++;
                        break;
                    }
                }
            }
            c[i]=count;
        }
        return c;
    }
}


2. Single Element in a Sorted Array (540)

Solution : class Solution {
    public int singleNonDuplicate(int[] nums) {
        Map<Integer, Integer> map = new HashMap<>();
        int n = nums.length;
        for (int i : nums) {
            map.put(i, map.getOrDefault(i, 0) + 1);
        }

        for (Map.Entry<Integer, Integer> entry : map.entrySet()) {
            if (entry.getValue() == 1) {
               int key = entry.getKey();
                return key;
            }
        }
        return -1;
    }
}

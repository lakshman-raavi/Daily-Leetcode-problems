1. Kth Missing Positive Number (1539)

Solution : class Solution {

    public int maxVal(int[] arr){
        int n=arr.length;
        int max=Integer.MIN_VALUE;
        for(int i=0;i<n;i++){
            if(max<arr[i]){
                max=arr[i];
            }
        }
        return max;
    }
    public int findKthPositive(int[] arr, int k) {
        int n=arr.length;
        int count=0;
        int max=maxVal(arr);
        int m=0;
        int res=0;
        for(int i=1;i<=max;i++){
            if(arr[m]!=i){
                k--;
                if(k==0){
                    return i;
                }
            }
            else
            m++;
        }
        if(k!=0){
            return max+k;
        }
        return -1;
    }
}

2. Check If N and Its Double Exist  (1346)

Solution : class Solution {
    public boolean checkIfExist(int[] arr) {
        int res=0;
        int n=arr.length;
        for(int i=0;i<n;i++){
            int num=arr[i];
            for(int j=0;j<n;j++){
                if(i==j){
                    continue;
                }
                if((arr[j]*2)==num){
                    return true;
                }
            }
        }
        return false;
    }
}


3. Keep Multiplying Found Values by Two  (2154)

Solution : class Solution {
    public int findFinalValue(int[] nums, int original) {
        List<Integer> list=new ArrayList<>();
        for(int i : nums){
            list.add(i);
        }

        for(int i : nums){
            if(list.contains(original)){
                original=2*original;
            }
        }
        return original;
    }
}

1. Find the Highest Altitude (1732)

Solution : class Solution {
    public int largestAltitude(int[] gain) {
        int max=Integer.MIN_VALUE;
        int sum=0;
        int n=gain.length;
        for(int i=0;i<n;i++){
            sum+=gain[i];
            if(sum>max){
                max=sum;
            }
        }
        return max<0?0:max;
    }
}

1.  Points That Intersect With Cars  (2848)

Solution :  class Solution {
    public int numberOfPoints(List<List<Integer>> nums) {
        Set<Integer> set = new HashSet<>();
        for (List<Integer> interval : nums) {
            int start = interval.get(0), end = interval.get(1);
            for(int i=start;i<=end;i++){
                set.add(i);
            }
        }
        return set.size();
    }
}


2. Merge Intervals  (56)

Solution :  class Solution {
    public int[][] merge(int[][] intervals) {
        Arrays.sort(intervals, (a, b) -> Integer.compare(a[0], b[0]));
        int n=intervals.length;
        int[][] res=new int[n][2];
        res[0]=intervals[0];
        int e=0;
        for(int i=1;i<n;i++){
            if(res[e][1]>=intervals[i][0]){
                res[e][1]=Math.max(res[e][1],intervals[i][1]);
            }
            else{
                e++;
                res[e]=intervals[i];
            }
        }

        return Arrays.copyOfRange(res, 0, e + 1);

    }
}


3.  Insert Interval  (57)

Solution :  class Solution {
    public int[][] insert(int[][] intervals, int[] newInterval) {
        int n1=intervals.length;
        int n2=newInterval.length;
        int[][] main=new int[n1+1][2];
        int e=0;
        for(int i=0;i<n1;i++){
            main[e]=intervals[i];
            e++;
        }
        main[e]=newInterval;
        e++;
        
        Arrays.sort(main, (a,b) -> Integer.compare(a[0],b[0]));
        int[][] res=new int[e][2];
        int k=0;
        res[0]=main[0];

        for(int i=1;i<e;i++){
            if(res[k][1]>=main[i][0]){
                res[k][1]=Math.max(res[k][1],main[i][1]);
            }
            else{
                k++;
                res[k]=main[i];
            }
        }

        return Arrays.copyOfRange(res,0,k+1);

    }
}

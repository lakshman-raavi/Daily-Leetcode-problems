1.  Koko Eating Bananas  (875)

Solution : class Solution {

    public int maxval(int[] a){
        int n=a.length;
        int max=Integer.MIN_VALUE;
        for(int i=0;i<n;i++){
            if(a[i]>max){
                max=a[i];
            }
        }
            return max;
        
    }
    public long calofhrs(int[] a,int mid){
        long res=0;
        int n=a.length;
        for(int i=0;i<n;i++){
            res+=(long)Math.ceil((double)a[i]/mid);
        }
        return res;
    }

    public int minEatingSpeed(int[] piles, int h) {
        int k = 1;
        int n = piles.length;
        int l=1,high=maxval(piles);
        int ans=0;
        while (l<=high) {
            long res = 0;
            int mid=l+(high-l)/2;
            res=calofhrs(piles,mid);
            if (res <= h) {
                ans=mid;
                high=mid-1;
            }
            else{
                l=mid+1;
            }
        }
        return ans;
    }
}

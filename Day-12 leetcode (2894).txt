1. Divisible and Non-divisible Sums Difference  (2894)

Solution : class Solution {
    public int differenceOfSums(int n, int m) {
        
        int res;
        int not=0,equal=0;
        for(int i=1;i<=n;i++){
            if(i%m!=0){
                not+=i;
            }
            else{
                equal+=i;
            }
        }

        return not-equal;
    }
}

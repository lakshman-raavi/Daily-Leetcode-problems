1. Search a 2D Matrix (74)
Solution : class Solution {
    public boolean searchMatrix(int[][] matrix, int target) {
        int n=matrix.length;
        int m=matrix[0].length;
        int l=0,high=m*n-1;
        int mid,row,col;
        while(l<=high){
            mid=l+(high-l)/2;
            row=mid/m;
            col=mid%m;

            if(matrix[row][col]==target){
                return true;
            }
            else if(matrix[row][col]<target){
                l=mid+1;
            }
            else{
                high=mid-1;
            }
        }
        return false;
    }
}

2. Rearrange Array Elements by Sign (2149)

Solution : class Solution {
    public int[] rearrangeArray(int[] nums) {
        int n=nums.length;
        int k1=0,k2=1;
        int[] res=new int[n];
        for(int i=0;i<n;i++){
            if(nums[i]>0){
                res[k1]=nums[i];
                k1+=2;
            }
            else{
                res[k2]=nums[i];
                k2+=2;
            }
        }
        return res;
    }
}

3. Replace Elements with Greatest Element on Right Side  (1299)

Solution :  class Solution {
    public int[] replaceElements(int[] arr) {
        for(int i=0;i<arr.length-1;i++){
            int max=0;
            for(int j=i+1;j<arr.length;j++){
                if(arr[j]>max){
                    max=arr[j];
                }
            }
            arr[i]=max;
        }
        arr[arr.length-1]=-1;
        return arr;
    }
}

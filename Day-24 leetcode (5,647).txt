1. Longest Palindromic Substring (5)

Solution : class Solution {

    public boolean funcal(String s){
        int l=0;
        int r=s.length()-1;
        while(l<r){
            if(s.charAt(l)!=s.charAt(r)){
                return false;
            }
            l++;
            r--;
        }
        return true;
    }

    public String longestPalindrome(String s) {
        int n=s.length();
        String str=new String();
        if(n == 0) 
        return "";
        int max=0;
        for(int i=0;i<n;i++){
            for(int j=i;j<n;j++){
                if(funcal(s.substring(i,j+1))){
                    int len=j-i+1;
                    if(len>max){
                        str=s.substring(i,j+1);
                        max=len;
                    }
                }
            }
        }
        return str;
    }
}

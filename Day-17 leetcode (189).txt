1. Rotate Array  (189)

Solution : class Solution {
    public void rotate(int[] nums, int k) {
        int n=nums.length;
        k=k%n;
        int[] temp=new int[k];
        int m=0;
        for(int i=n-k;i<n;i++){
            temp[m]=nums[i];
            m++;
        }
        int[] t2=new int[n-k];
        int l=0;
        for(int i=0;i<n-k;i++){
            t2[l]=nums[i];
            l++;
        }
        for(int i=0;i<k;i++){
            nums[i]=temp[i];
        }

        for(int i=k;i<n;i++){
            nums[i]=t2[i-k];
        }
       
    }
}

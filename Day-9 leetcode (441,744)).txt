1. Arranging Coins  (441)

Solution : class Solution {
    public int arrangeCoins(int n) {
        int count=0;
        for(int i=0;i<n;i++){
            if(n<0){
                break;
            }
            n=n-(i+1);
            count++;
        }
        return count;
    }
}


2. Find Smallest Letter Greater Than Target (744)

Solution : class Solution {
    public char nextGreatestLetter(char[] letters, char target) {
        int tar=(int)target;
        int mini=Integer.MAX_VALUE;
        for(int i=0;i<letters.length;i++){
                int ascii=(int)letters[i];
                if(ascii>target){
                    mini=Math.min(mini,ascii);
                }
        }
        if(mini==Integer.MAX_VALUE){
            return letters[0];
        }
        return (char)mini;
    }
}

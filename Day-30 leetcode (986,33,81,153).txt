1. Interval List Intersections  (986)

Solution :  class Solution {
    public int[][] intervalIntersection(int[][] firstList, int[][] secondList) {
        int n1=firstList.length;
        int n2=secondList.length;
        
        if(n1==0 || n2==0){
            return new int[0][2];
        }
        int[][] res=new int[n1+n2][2];
        int e=0;
        int i=0;
        int j=0;
        while(i<n1 && j<n2){
            int start = Math.max(firstList[i][0], secondList[j][0]);
            int end = Math.min(firstList[i][1], secondList[j][1]);

            if (start <= end) {
                res[e][0] = start;
                res[e][1] = end;
                e++;
            }
            if (firstList[i][1] < secondList[j][1]) {
                i++;
            } else {
                j++;
            }
        }

        return Arrays.copyOfRange(res,0,e);
    }
}


2.  Search in Rotated Sorted Array  (33)

Solution :  class Solution {
    public int search(int[] nums, int target) {
        
        // for(int i=0;i<nums.length;i++){
        //     if(nums[i]==target){
        //         return i;
        //     }
        // }
        // return -1;


        int low=0;
        int high=nums.length-1;
        while(low<=high){
            int mid=low+(high-low)/2;
            if(nums[mid]==target){
                return mid;
            }

            //left part
            if(nums[low]<=nums[mid]){
            if(nums[low]<=target && target<=nums[mid]){
                high=mid-1;
            }
            else{
                low=mid+1;
            }
            }

            //right part
            else{
                if(nums[mid]<=target && target<=nums[high]){
                    low=mid+1;
                }
                else{
                    high=mid-1;
                }
            }
        }

        return -1;
    }
}


3.  Search in Rotated Sorted Array II  (81)

Solution  :  class Solution {
    public boolean search(int[] nums, int target) {
        
        // for(int i=0;i<nums.length;i++){
        //     if(nums[i]==target){
        //         return true;
        //     }
        // }
        // return false;


        int low=0;
        int high=nums.length-1;
        while(low<=high){
            int mid=low+(high-low)/2;
            if(nums[mid]==target){
                return true;
            }

            if(nums[low]==nums[mid] && nums[mid]==nums[high]){
                low++;
                high--;
                continue;
            } 

            //left part
            if(nums[low]<=nums[mid]){
            if(nums[low]<=target && target<=nums[mid]){
                high=mid-1;
            }
            else{
                low=mid+1;
            }
            }

            //right part
            else{
                if(nums[mid]<=target && target<=nums[high]){
                    low=mid+1;
                }
                else{
                    high=mid-1;
                }
            }
        }

        return false;

    }
}


4. Find Minimum in Rotated Sorted Array  (153)

Solution :  class Solution {
    public int findMin(int[] nums) {
        // int min=Integer.MAX_VALUE;
        // for(int i=0;i<nums.length;i++){
        //     if(nums[i]<min){
        //         min=nums[i];
        //     }
        // }

        // return min;


        int low=0;
        int high=nums.length-1;
        int min=Integer.MAX_VALUE;
        while(low<=high){
            int mid=(low+high)/2;
            if(nums[mid]<min){
                min=nums[mid];
            }

            //right part
            if(nums[mid]>nums[high]){
                low=mid+1;
            }
            else{
                high=mid-1;
            }
            
        }
        return min;
    }
}

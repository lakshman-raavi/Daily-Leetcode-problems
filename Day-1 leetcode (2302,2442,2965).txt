1.Count Subarrays With Score Less Than K  (2302)
Solution:

class Solution {
    public long countSubarrays(int[] nums, long k) {
        long n = nums.length;
        long count = 0, sum = 0;
        int l = 0, r;
        for (r = 0; r < n; r++) {
            sum += nums[r];
            while (sum * (r - l + 1) >= k) {
                sum -= nums[l];
                l++;
            }
            count += (r - l + 1);

        }

        return count;
    }
}


2.Count Number of Distinct Integers After Reverse Operations (2442)
Solution:  
class Solution {

    public int reversenum(int n){
        int res=0;
        while(n!=0){
            res=res*10+(n%10);
            n/=10;
        }
        return res;
    }
    public int countDistinctIntegers(int[] nums) {

        int n=nums.length;
        Set<Integer> set=new HashSet<>();
        for(int x: nums){
            set.add(x);
        }

        for(int i=0;i<n;i++){
            set.add(reversenum(nums[i]));
        }
        return set.size();
    }
}

3. Find Missing and Repeated Values (2965)

class Solution {
    public int[] findMissingAndRepeatedValues(int[][] grid) {
        int n=grid.length;
        int m=grid[0].length;
        Set<Integer> set=new HashSet<>();
        int[] res=new int[2];
         for (int[] row : grid) {
            for (int x : row) {
               if(!set.contains(x)){
                set.add(x);
               }
               else{
                res[0]=x;
               }
            }
        }

        int t=n*n;
        for(int i=1;i<=t;i++){
            if(!set.contains(i)){
                res[1]=i;
                break;
            }
        }
    return res;
        
    }
}
